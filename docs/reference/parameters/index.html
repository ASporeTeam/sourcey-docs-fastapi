Using them

from typing import Annotated

from fastapi import Depends, FastAPI, Query

app = FastAPI()


def api_key_header(x_key: Annotated[str, Query()]):
    return x_key


@app.get("/items/")
async def read_items(
    q: Annotated[str | None, Query(max_length=50)] = None,
    x_key: Annotated[str, Depends(api_key_header)] = "",
):
    return {"q": q, "x_key": x_key}

Every helper above is defined in fastapi.param_functions and is re-exported from the package root, so from fastapi import Depends, Query is the public path.
